# Ostrich input: calibrate aquifer parameters of the Raven-MODFLOW 6 coupling against head observations.
# Layout: this folder holds ostIn.txt, Ost-Raven.sh and Liard.rvp.tpl; ./model/ holds the Raven model
# (all input files, data folders, and :MF6Library pointing to libmf6).
ProgramType         DDS
ObjectiveFunction   GCOP
ModelExecutable     ./Ost-Raven.sh
PreserveBestModel   ./save_best.sh

BeginFilePairs
  Liard.rvp.tpl ; model/Liard.rvp
EndFilePairs

# hydraulic conductivity is calibrated in log space
BeginParams
  #name          init    low     high    tx_in  tx_ost  tx_out
  par_K_gravel   5.0     1.0     500.0   none   log10   none
  par_Sy_gravel  0.10    0.05    0.35    none   none    none
  par_K_till     2.0     0.01    5.0     none   log10   none
EndParams

# one response per monitoring well: RMSE of GW_HEAD in Raven's Diagnostics.csv. Ostrich counts lines from 0 (the header
# is line 0) and columns from 1. With :EvaluationMetrics NASH_SUTCLIFFE RMSE KLING_GUPTA the columns are
# observed_data_series, filename, DIAG_NASH_SUTCLIFFE, DIAG_RMSE (column 4), DIAG_KLING_GUPTA. A series' line is its
# place among the observations: lines 1-4 when the four GW_HEAD series are the model's only observations (as here); with
# four monitoring wells added to the Liard example, whose 29 streamflow series come first, they are lines 30-33. Check both
# against your Diagnostics.csv.
BeginResponseVars
  #name   filename                        keyword   line  col  token
  RMSE1   model/output/Diagnostics.csv ;  OST_NULL  1     4    ','
  RMSE2   model/output/Diagnostics.csv ;  OST_NULL  2     4    ','
  RMSE3   model/output/Diagnostics.csv ;  OST_NULL  3     4    ','
  RMSE4   model/output/Diagnostics.csv ;  OST_NULL  4     4    ','
EndResponseVars

BeginTiedRespVars
  SUM_RMSE 4 RMSE1 RMSE2 RMSE3 RMSE4 wsum 1.0 1.0 1.0 1.0
EndTiedRespVars

BeginGCOP
  CostFunction SUM_RMSE
  PenaltyFunction APM
EndGCOP

BeginConstraints
EndConstraints

RandomSeed 11

BeginDDSAlg
  PerturbationValue 0.20
  MaxIterations     150
  UseInitialParamValues
EndDDSAlg
